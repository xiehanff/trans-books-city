# C 标准库：字符串、输入输出与内存函数

C 标准库提供了大量实用函数。掌握这些函数能大幅提升开发效率，避免重复造轮子。

## stdio.h：输入输出

### printf 格式化输出

```c
#include <stdio.h>

int main(void) {
    int i = 42;
    double d = 3.14159;
    char c = 'A';
    char *s = "hello";

    printf("整数: %d\n", i);        // 42
    printf("十六进制: %x\n", i);     // 2a
    printf("八进制: %o\n", i);       // 52
    printf("浮点数: %f\n", d);       // 3.141590
    printf("科学计数: %e\n", d);     // 3.141590e+00
    printf("字符: %c\n", c);        // A
    printf("字符串: %s\n", s);       // hello
    printf("指针: %p\n", (void*)&i); // 地址值

    return 0;
}
```

常用格式说明符：

| 格式 | 说明 |
|-----|------|
| `%d` | 十进制整数 |
| `%x` | 十六进制（小写） |
| `%X` | 十六进制（大写） |
| `%o` | 八进制 |
| `%u` | 无符号整数 |
| `%ld` | long 类型 |
| `%lld` | long long 类型 |
| `%f` | 浮点数，`printf` 中 `double` 也用这个格式 |
| `%e` | 科学计数法 |
| `%g` | 自动选择 %f 或 %e |
| `%c` | 字符 |
| `%s` | 字符串 |
| `%p` | 指针 |
| `%%` | 百分号 |

宽度和精度控制：

```c
printf("%5d\n", 42);      // "   42"，宽度 5，右对齐
printf("%-5d\n", 42);     // "42   "，左对齐
printf("%05d\n", 42);     // "00042"，用 0 填充
printf("%.2f\n", 3.14159); // "3.14"，保留 2 位小数
printf("%8.2f\n", 3.14);  // "    3.14"，总宽度 8，小数 2 位
printf("%10.4s\n", "hello"); // "    hell"，字符串截取
```

### scanf 格式化输入

```c
int i;
double d;
char s[100];

scanf("%d", &i);     // 读取整数，注意 &
scanf("%lf", &d);    // 读取 double
scanf("%s", s);      // 读取字符串，遇空白停止，不需 &
scanf("%99s", s);    // 限制最大长度，防止溢出
```

`scanf` 返回成功读取的项数：

```c
int a, b;
int result = scanf("%d %d", &a, &b);
if (result != 2) {
    printf("输入格式错误\n");
}
```

读取带空格的字符串：

```c
char line[100];
fgets(line, sizeof(line), stdin);  // 读取一行，包括空格
```

### 文件操作

#### 打开与关闭

```c
FILE *fopen(const char *filename, const char *mode);

// 模式：
// "r"  - 只读，文件必须存在
// "w"  - 只写，文件不存在则创建，存在则清空
// "a"  - 追加，文件不存在则创建
// "r+" - 读写，文件必须存在
// "w+" - 读写，文件不存在则创建，存在则清空
// "a+" - 读和追加
// "rb", "wb" - 二进制模式

FILE *fp = fopen("data.txt", "r");
if (fp == NULL) {
    perror("打开文件失败");
    return -1;
}

// ... 操作文件 ...

fclose(fp);  // 关闭文件
```

#### 字符与行操作

```c
// 读取一个字符
int ch = fgetc(fp);
putchar(ch);

// 写入一个字符
fputc('A', fp);

// 读取一行
char line[256];
if (fgets(line, sizeof(line), fp) != NULL) {
    printf("%s", line);
}

// 写入一行
fputs("hello world\n", fp);
```

#### 块读写

```c
// 二进制块读写
size_t fread(void *ptr, size_t size, size_t count, FILE *stream);
size_t fwrite(const void *ptr, size_t size, size_t count, FILE *stream);

// 写入数组
int arr[] = {1, 2, 3, 4, 5};
fwrite(arr, sizeof(int), 5, fp);

// 读取数组
int buf[5];
size_t n = fread(buf, sizeof(int), 5, fp);
// n 是实际读取的元素个数
```

二进制文件读写示例：

```c
#include <stdio.h>

struct Person {
    char name[32];
    int age;
};

int main(void) {
    // 写入
    FILE *fp = fopen("people.dat", "wb");
    if (!fp) return -1;

    struct Person people[] = {
        {"Alice", 25},
        {"Bob", 30}
    };
    fwrite(people, sizeof(struct Person), 2, fp);
    fclose(fp);

    // 读取
    fp = fopen("people.dat", "rb");
    if (!fp) return -1;

    struct Person p;
    while (fread(&p, sizeof(struct Person), 1, fp) == 1) {
        printf("%s: %d\n", p.name, p.age);
    }
    fclose(fp);

    return 0;
}
```

#### 文件定位

```c
// 移动文件指针
fseek(fp, 0, SEEK_SET);   // 文件开头
fseek(fp, 0, SEEK_END);   // 文件末尾
fseek(fp, -10, SEEK_CUR); // 当前位置前移 10 字节

// 获取当前位置
long pos = ftell(fp);

// 重置到开头
rewind(fp);
```

#### 格式化文件读写

```c
fprintf(fp, "Name: %s, Age: %d\n", "Alice", 25);

int age;
char name[32];
fscanf(fp, "Name: %s, Age: %d", name, &age);
```

## stdlib.h：内存管理

### malloc 动态分配

```c
void *malloc(size_t size);

// 分配 10 个 int 的空间
int *arr = malloc(10 * sizeof(int));
if (arr == NULL) {
    printf("内存分配失败\n");
    return -1;
}

// 使用
for (int i = 0; i < 10; i++) {
    arr[i] = i * 2;
}

// 释放
free(arr);
arr = NULL;  // 避免悬空指针
```

`malloc` 分配的内存不会初始化，内容是随机的。

### calloc 分配并初始化

```c
void *calloc(size_t num, size_t size);

// 分配 10 个 int 并初始化为 0
int *arr = calloc(10, sizeof(int));
// 等价于
int *arr = malloc(10 * sizeof(int));
memset(arr, 0, 10 * sizeof(int));
```

### realloc 调整大小

```c
void *realloc(void *ptr, size_t new_size);

int *arr = malloc(10 * sizeof(int));

// 扩展到 20 个 int
int *temp = realloc(arr, 20 * sizeof(int));
if (temp == NULL) {
    // 分配失败，原内存仍然有效
    free(arr);
    return -1;
}
arr = temp;  // 使用新指针

// 缩小到 5 个 int
temp = realloc(arr, 5 * sizeof(int));
if (temp) arr = temp;
```

`realloc` 行为：

- `ptr` 为 NULL 时等价于 `malloc`
- `new_size` 为 0 时等价于 `free`
- 可能返回新地址，原数据自动复制
- 可能返回原地址（原地扩展）

### free 释放内存

```c
void free(void *ptr);

int *p = malloc(sizeof(int));
*p = 42;
free(p);    // 释放内存
p = NULL;   // 指针置空

// 注意
free(NULL);  // 安全，什么都不做
int a;
free(&a);    // 错误！不能释放栈内存
free(p);     // 重复释放同一指针 - 错误！
```

内存泄漏示例：

```c
void leak(void) {
    int *p = malloc(100);
    // 忘记 free(p)，内存泄漏
}

// 另一种泄漏：丢失指针
int *p = malloc(100);
p = malloc(200);  // 第一个 malloc 的内存无法释放了
```

### 内存管理原则

1. 谁分配，谁释放
2. 分配后检查是否成功
3. 释放后指针置空
4. 避免重复释放
5. 避免内存泄漏

## string.h：字符串函数

### strlen 字符串长度

```c
size_t strlen(const char *s);

char *s = "hello";
printf("%zu\n", strlen(s));  // 5，不包括 '\0'

// 注意与 sizeof 的区别
char str[] = "hello";
printf("%zu\n", strlen(str));  // 5
printf("%zu\n", sizeof(str));  // 6，包括 '\0'
```

### strcpy / strncpy 字符串复制

```c
char *strcpy(char *dest, const char *src);
char *strncpy(char *dest, const char *src, size_t n);

char dest[100];
strcpy(dest, "hello");  // 复制整个字符串

strncpy(dest, "hello", 3);  // 只复制 3 个字符
dest[3] = '\0';  // strncpy 不保证以 '\0' 结尾，需手动添加
```

安全的复制方式：

```c
char dest[10];
strncpy(dest, src, sizeof(dest) - 1);
dest[sizeof(dest) - 1] = '\0';  // 确保结尾
```

### strcat / strncat 字符串连接

```c
char *strcat(char *dest, const char *src);
char *strncat(char *dest, const char *src, size_t n);

char s[50] = "hello";
strcat(s, " world");  // s = "hello world"

// 安全连接
strncat(s, append, sizeof(s) - strlen(s) - 1);
```

`strncat` 会自动添加 `\0`，这是与 `strncpy` 的区别。

### strcmp / strncmp 字符串比较

```c
int strcmp(const char *s1, const char *s2);
int strncmp(const char *s1, const char *s2, size_t n);

// 返回值：
// < 0 表示 s1 < s2
// = 0 表示 s1 == s2
// > 0 表示 s1 > s2

if (strcmp(s1, s2) == 0) {
    printf("字符串相等\n");
}

// 只比较前 n 个字符
if (strncmp(s1, s2, 3) == 0) {
    printf("前 3 个字符相同\n");
}
```

字典序比较基于 ASCII 码：

```c
strcmp("abc", "abd");   // < 0
strcmp("abc", "abc");   // 0
strcmp("abd", "abc");   // > 0
strcmp("ABC", "abc");   // < 0，大写字母 ASCII 码更小
```

### strchr / strrchr 字符查找

```c
char *strchr(const char *s, int c);   // 从头找
char *strrchr(const char *s, int c);  // 从尾找

char *s = "hello world";
char *p = strchr(s, 'o');     // 指向第一个 'o'
char *q = strrchr(s, 'o');    // 指向最后一个 'o'

if (p) {
    printf("找到: %s\n", p);  // "o world"
}

// 未找到返回 NULL
if (strchr(s, 'x') == NULL) {
    printf("未找到\n");
}
```

### strstr 字符串查找

```c
char *strstr(const char *haystack, const char *needle);

char *s = "hello world";
char *p = strstr(s, "wor");  // 指向 "world"

if (p) {
    printf("找到: %s\n", p);
}
```

### strtok 字符串分割

```c
char *strtok(char *str, const char *delim);

char str[] = "one,two,three";
char *token = strtok(str, ",");  // 首次调用传入字符串

while (token != NULL) {
    printf("%s\n", token);
    token = strtok(NULL, ",");   // 后续调用传入 NULL
}

// 输出:
// one
// two
// three
```

`strtok` 会修改原字符串，把分隔符替换为 `\0`。多线程环境下不安全，可使用 `strtok_r`。

## string.h：内存操作函数

### memcpy 内存复制

```c
void *memcpy(void *dest, const void *src, size_t n);

int a[] = {1, 2, 3, 4, 5};
int b[5];
memcpy(b, a, sizeof(a));  // 复制整个数组

// 复制结构体
struct Point { int x, y; } p1 = {10, 20}, p2;
memcpy(&p2, &p1, sizeof(struct Point));
```

`memcpy` 不处理源和目标重叠的情况，重叠时行为未定义。

### memmove 内存移动

```c
void *memmove(void *dest, const void *src, size_t n);

// memmove 可以正确处理重叠
char str[] = "hello world";
memmove(str + 6, str, 5);  // 将 "hello" 移到 "world" 位置
printf("%s\n", str);  // "hello hello"
```

源和目标有重叠时用 `memmove`，无重叠时用 `memcpy`（更快）。

图示说明：

```
内存重叠情况：
源区域:    [A B C D E]
目标区域:      [A B C D E]
           ↑ 重叠部分

memcpy：  直接复制可能出错（重叠部分源数据被覆盖）
memmove： 先复制到临时区域再写入，保证正确
```

### memset 内存设置

```c
void *memset(void *s, int c, size_t n);

// 数组清零
int arr[100];
memset(arr, 0, sizeof(arr));

// 字符数组填充
char str[100];
memset(str, 'A', 99);
str[99] = '\0';

// 常见用法：结构体初始化
struct Config cfg;
memset(&cfg, 0, sizeof(cfg));
```

注意：`memset` 按字节填充，不适合非 0 的 int 数组初始化：

```c
int arr[100];
memset(arr, 1, sizeof(arr));  // 每个 int 变成 0x01010101，不是 1
```

### memcmp 内存比较

```c
int memcmp(const void *s1, const void *s2, size_t n);

int a[] = {1, 2, 3};
int b[] = {1, 2, 4};

int result = memcmp(a, b, sizeof(a));
// result < 0，因为 3 < 4

// 比较结构体
struct Point p1 = {1, 2}, p2 = {1, 3};
if (memcmp(&p1, &p2, sizeof(struct Point)) == 0) {
    printf("相同\n");
}
```

## 实际应用示例

### 示例 1：读取文件到字符串

```c
#include <stdio.h>
#include <stdlib.h>

char* read_file(const char *filename) {
    FILE *fp = fopen(filename, "r");
    if (!fp) return NULL;

    // 获取文件大小
    fseek(fp, 0, SEEK_END);
    long size = ftell(fp);
    rewind(fp);

    // 分配内存
    char *content = malloc(size + 1);
    if (!content) {
        fclose(fp);
        return NULL;
    }

    // 读取内容
    size_t read = fread(content, 1, size, fp);
    content[read] = '\0';

    fclose(fp);
    return content;
}

// 使用
char *content = read_file("data.txt");
if (content) {
    printf("%s\n", content);
    free(content);
}
```

### 示例 2：动态字符串拼接

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdarg.h>
#include <string.h>

char* str_join(const char *sep, int count, ...) {
    va_list ap;
    size_t total_len = 0;
    size_t sep_len = strlen(sep);

    // 第一遍：计算总长度
    va_start(ap, count);
    for (int i = 0; i < count; i++) {
        const char *s = va_arg(ap, const char*);
        total_len += strlen(s);
        if (i > 0) total_len += sep_len;
    }
    va_end(ap);

    // 分配内存
    char *result = malloc(total_len + 1);
    if (!result) return NULL;
    result[0] = '\0';

    // 第二遍：拼接字符串
    va_start(ap, count);
    for (int i = 0; i < count; i++) {
        const char *s = va_arg(ap, const char*);
        if (i > 0) strcat(result, sep);
        strcat(result, s);
    }
    va_end(ap);

    return result;
}

// 使用
char *s = str_join(", ", 3, "apple", "banana", "cherry");
printf("%s\n", s);  // "apple, banana, cherry"
free(s);
```

### 示例 3：简单的字符串数组

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    char **items;
    int count;
    int capacity;
} StringArray;

StringArray* str_array_new(int capacity) {
    StringArray *arr = malloc(sizeof(StringArray));
    if (!arr) return NULL;
    arr->items = calloc(capacity, sizeof(char*));
    if (!arr->items) {
        free(arr);
        return NULL;
    }
    arr->count = 0;
    arr->capacity = capacity;
    return arr;
}

void str_array_free(StringArray *arr) {
    if (!arr) return;
    for (int i = 0; i < arr->count; i++) {
        free(arr->items[i]);
    }
    free(arr->items);
    free(arr);
}

int str_array_append(StringArray *arr, const char *s) {
    if (arr->count >= arr->capacity) {
        int new_cap = arr->capacity * 2;
        char **new_items = realloc(arr->items, new_cap * sizeof(char*));
        if (!new_items) return -1;
        arr->items = new_items;
        arr->capacity = new_cap;
    }

    arr->items[arr->count] = malloc(strlen(s) + 1);
    if (!arr->items[arr->count]) return -1;
    strcpy(arr->items[arr->count], s);
    arr->count++;
    return 0;
}

// 使用
StringArray *arr = str_array_new(4);
str_array_append(arr, "hello");
str_array_append(arr, "world");

for (int i = 0; i < arr->count; i++) {
    printf("%s\n", arr->items[i]);
}

str_array_free(arr);
```

### 示例 4：分割字符串到数组

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

char** str_split(const char *s, const char *delim, int *count) {
    char *copy = malloc(strlen(s) + 1);
    if (!copy) return NULL;
    strcpy(copy, s);

    // 计算分割后的数量
    int n = 0;
    char *token = strtok(copy, delim);
    while (token) {
        n++;
        token = strtok(NULL, delim);
    }

    // 分配指针数组
    char **result = malloc((n + 1) * sizeof(char*));
    if (!result) {
        free(copy);
        return NULL;
    }

    // 重新分割并复制
    strcpy(copy, s);
    int i = 0;
    token = strtok(copy, delim);
    while (token) {
        result[i] = malloc(strlen(token) + 1);
        strcpy(result[i], token);
        i++;
        token = strtok(NULL, delim);
    }
    result[i] = NULL;
    *count = n;

    free(copy);
    return result;
}

void free_split(char **arr) {
    if (!arr) return;
    for (int i = 0; arr[i]; i++) {
        free(arr[i]);
    }
    free(arr);
}

// 使用
int count;
char **parts = str_split("one,two,three", ",", &count);
for (int i = 0; i < count; i++) {
    printf("%s\n", parts[i]);
}
free_split(parts);
```
