# 函数接口

函数是 C 语言程序的基本组成单位。理解函数参数传递机制、返回值设计以及各种参数类型的使用方式，是写出高质量 C 代码的基础。

## 函数参数与返回值

### 参数传递机制

C 语言函数参数传递只有一种方式：值传递。

```c
void increment(int x) {
    x++;
}

int a = 10;
increment(a);
printf("%d\n", a);  // 10，没有变化
```

`increment(a)` 调用时，`a` 的值被复制给参数 `x`。函数内部修改的是 `x`，与 `a` 无关。

### 返回值

函数用 `return` 返回结果给调用者：

```c
int add(int a, int b) {
    return a + b;
}

int sum = add(3, 5);  // sum = 8
```

不返回值得函数声明为 `void`：

```c
void print_hello(void) {
    printf("Hello!\n");
    // 没有 return，或者只写 return;
}
```

不要返回局部变量的指针：

```c
int* wrong(void) {
    int x = 10;
    return &x;  // 错误！x 在函数结束后销毁
}

int* right(void) {
    int *p = malloc(sizeof(int));
    *p = 10;
    return p;  // 正确，堆内存需要调用者释放
}
```

返回静态变量或字符串字面量是安全的：

```c
char* get_message(void) {
    return "Hello";  // 正确，字符串字面量在常量区
}

int* get_counter(void) {
    static int count = 0;
    count++;
    return &count;  // 正确，静态变量不会销毁
}
```

## 指针作为参数

要修改调用者的变量，需要传递指针：

```c
void increment(int *p) {
    (*p)++;
}

int a = 10;
increment(&a);
printf("%d\n", a);  // 11
```

### 交换两个变量

```c
void swap(int *x, int *y) {
    int temp = *x;
    *x = *y;
    *y = temp;
}

int a = 1, b = 2;
swap(&a, &b);
printf("a=%d, b=%d\n", a, b);  // a=2, b=1
```

### 指针作为输出参数

函数通过指针参数"返回"多个值：

```c
// 计算商和余数
void divide(int a, int b, int *quotient, int *remainder) {
    *quotient = a / b;
    *remainder = a % b;
}

int q, r;
divide(17, 5, &q, &r);
printf("商=%d, 余数=%d\n", q, r);  // 商=3, 余数=2
```

### 通过指针实现"引用传递"

```c
void allocate_array(int **pp, int n) {
    *pp = malloc(n * sizeof(int));
}

int *arr = NULL;
allocate_array(&arr, 10);
// arr 现在指向分配的内存
free(arr);
```

## 数组作为参数

数组作为参数时，实际上传递的是指针：

```c
// 以下三种声明等价
int sum(int arr[], int n);
int sum(int arr[10], int n);  // 10 会被忽略
int sum(int *arr, int n);
```

必须额外传递数组长度，因为函数内部无法知道数组大小：

```c
int sum(int *arr, int n) {
    int total = 0;
    for (int i = 0; i < n; i++) {
        total += arr[i];
    }
    return total;
}

int a[] = {1, 2, 3, 4, 5};
int total = sum(a, 5);
```

### 多维数组作为参数

必须指定除第一维之外的所有维度：

```c
// 正确
void print_matrix(int m[][3], int rows) {
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < 3; j++) {
            printf("%d ", m[i][j]);
        }
        printf("\n");
    }
}

// 更通用的写法：传递指针，手动计算偏移
void print_matrix_generic(int *m, int rows, int cols) {
    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            printf("%d ", m[i * cols + j]);
        }
        printf("\n");
    }
}

int matrix[2][3] = {{1, 2, 3}, {4, 5, 6}};
print_matrix(matrix, 2);
print_matrix_generic(&matrix[0][0], 2, 3);
```

### 字符串作为参数

字符串本质是 `char*`，可以作为参数：

```c
int str_len(const char *s) {
    int len = 0;
    while (*s++) {
        len++;
    }
    return len;
}

char *str = "hello";
printf("%d\n", str_len(str));  // 5
```

`const` 表示函数不会修改字符串内容。

## 结构体作为参数

### 结构体值传递

直接传递结构体会复制整个结构体：

```c
struct Point {
    int x;
    int y;
};

void print_point(struct Point p) {
    printf("(%d, %d)\n", p.x, p.y);
}

struct Point pt = {10, 20};
print_point(pt);  // 复制整个结构体
```

小结构体可以值传递，大结构体应传递指针避免复制开销：

```c
void print_point_ptr(const struct Point *p) {
    printf("(%d, %d)\n", p->x, p->y);
}

struct Point pt = {10, 20};
print_point_ptr(&pt);  // 只传递指针
```

### 修改结构体

```c
void move_point(struct Point *p, int dx, int dy) {
    p->x += dx;
    p->y += dy;
}

struct Point pt = {10, 20};
move_point(&pt, 5, -3);
printf("(%d, %d)\n", pt.x, pt.y);  // (15, 17)
```

### 返回结构体

C 语言可以直接返回结构体：

```c
struct Point create_point(int x, int y) {
    struct Point p = {x, y};
    return p;
}

struct Point pt = create_point(10, 20);
```

也可以返回结构体指针（通常指向动态分配的内存）：

```c
struct Point* alloc_point(int x, int y) {
    struct Point *p = malloc(sizeof(struct Point));
    p->x = x;
    p->y = y;
    return p;
}

struct Point *pt = alloc_point(10, 20);
free(pt);
```

## 回调函数

回调函数是通过函数指针传递给其他函数的函数，用于实现"控制反转"。

### 基本用法

```c
// 定义函数指针类型
typedef int (*CompareFunc)(int a, int b);

// 在数组中查找元素
int find(int *arr, int n, int target, CompareFunc cmp) {
    for (int i = 0; i < n; i++) {
        if (cmp(arr[i], target) == 0) {
            return i;
        }
    }
    return -1;
}

// 比较函数
int int_equal(int a, int b) {
    return a - b;
}

int arr[] = {10, 20, 30, 40};
int index = find(arr, 4, 30, int_equal);  // 2
```

### 标准库中的回调

`qsort` 使用回调进行比较：

```c
#include <stdlib.h>

int compare_int(const void *a, const void *b) {
    return *(int*)a - *(int*)b;
}

int arr[] = {5, 2, 8, 1, 9};
qsort(arr, 5, sizeof(int), compare_int);
// arr 变为 {1, 2, 5, 8, 9}
```

`bsearch` 使用回调进行二分查找：

```c
int key = 5;
int *result = bsearch(&key, arr, 5, sizeof(int), compare_int);
if (result != NULL) {
    printf("找到: %d\n", *result);
}
```

### 带上下文的回调

C 语言没有闭包，需要手动传递上下文：

```c
typedef int (*CompareContext)(const void *a, const void *b, void *context);

int compare_with_context(const void *a, const void *b, void *context) {
    int *weights = (int*)context;
    int idx_a = *(int*)a;
    int idx_b = *(int*)b;
    return weights[idx_a] - weights[idx_b];
}
```

## 可变参数函数

函数可以接受不定数量的参数，`printf` 就是最典型的例子。

### 基本用法

使用 `stdarg.h` 中的宏：

```c
#include <stdarg.h>
#include <stdio.h>

// 计算任意数量整数的平均值
double average(int count, ...) {
    va_list ap;
    double sum = 0;

    va_start(ap, count);  // 初始化，count 是最后一个固定参数

    for (int i = 0; i < count; i++) {
        sum += va_arg(ap, int);  // 逐个获取参数
    }

    va_end(ap);  // 清理
    return sum / count;
}

double avg = average(4, 10, 20, 30, 40);  // 25.0
```

### 可变参数宏

| 宏 | 作用 |
|---|------|
| `va_list` | 声明可变参数列表变量 |
| `va_start(ap, last)` | 初始化，`last` 是最后一个固定参数 |
| `va_arg(ap, type)` | 获取下一个参数，指定类型 |
| `va_end(ap)` | 清理 |

### 实现简化版 printf

```c
void my_printf(const char *format, ...) {
    va_list ap;
    va_start(ap, format);

    while (*format) {
        if (*format == '%') {
            format++;
            switch (*format) {
                case 'd': {
                    int val = va_arg(ap, int);
                    printf("%d", val);
                    break;
                }
                case 's': {
                    char *val = va_arg(ap, char*);
                    printf("%s", val);
                    break;
                }
                case 'c': {
                    // 注意：char 在可变参数中提升为 int
                    int val = va_arg(ap, int);
                    putchar(val);
                    break;
                }
                case '%':
                    putchar('%');
                    break;
                default:
                    putchar('%');
                    putchar(*format);
            }
        } else {
            putchar(*format);
        }
        format++;
    }

    va_end(ap);
}
```

### 注意事项

1. 可变参数至少需要一个固定参数（用于定位）

```c
void wrong(...) { }  // 错误！
void right(int first, ...) { }  // 正确
```

2. `va_arg` 的类型必须与实际传入类型匹配

```c
double avg = average(3, 1.5, 2.5, 3.5);  // 错误！double 和 int 不匹配
```

3. 函数内部无法知道传入参数的数量，需要约定传递方式

```c
// 方式一：固定参数指定数量
printf_args(int count, ...);

// 方式二：哨兵值标记结束
printf_args_end(int first, ...);  // 假设以 0 结尾

// 方式三：格式字符串指定
printf(const char *fmt, ...);
```

4. 默认参数提升

在可变参数中，`char` 和 `short` 会提升为 `int`，`float` 会提升为 `double`：

```c
char c = 'A';
float f = 3.14f;

some_varargs(c, f);  // 实际传入 int 和 double

va_arg(ap, int);     // 正确
va_arg(ap, char);    // 错误！
va_arg(ap, double);  // 正确
va_arg(ap, float);   // 错误！
```

## 函数设计最佳实践

### 单一职责

一个函数只做一件事：

```c
// 不好：一个函数做了太多事
void process_file(const char *filename) {
    FILE *f = fopen(filename, "r");
    // 读取
    // 解析
    // 处理
    // 写入结果
    fclose(f);
}

// 好：拆分职责
char* read_file(const char *filename);
Data parse_content(const char *content);
Result process_data(Data *data);
void write_result(const char *filename, Result *result);
```

### 参数设计

控制参数数量，超过 4 个考虑封装：

```c
// 参数太多
void draw_rect(int x, int y, int w, int h,
               int color, int border, int shadow, int opacity);

// 封装为结构体
struct RectStyle {
    int x, y, w, h;
    int color;
    int border;
    int shadow;
    int opacity;
};

void draw_rect(const struct RectStyle *style);
```

### 错误处理

使用返回值或输出参数表示错误：

```c
// 返回值表示成功/失败
int divide(int a, int b, int *result) {
    if (b == 0) {
        return -1;  // 错误码
    }
    *result = a / b;
    return 0;  // 成功
}

int r;
if (divide(10, 0, &r) != 0) {
    printf("除零错误\n");
}
```

### const 修饰

不修改的参数用 `const` 保护：

```c
// 字符串处理函数的标准写法
size_t my_strlen(const char *s);
char* my_strcpy(char *dest, const char *src);
int my_strcmp(const char *s1, const char *s2);
```

### 避免全局变量

```c
// 不好：使用全局变量
int g_total;

void add_to_total(int n) {
    g_total += n;
}

// 好：通过参数传递
void add_to_total(int *total, int n) {
    *total += n;
}
```

### 函数原型

复杂函数添加原型声明在头文件，便于复用：

```c
// utils.h
#ifndef UTILS_H
#define UTILS_H

int sum_array(const int *arr, int n);
void swap_int(int *a, int *b);
char* duplicate_string(const char *s);

#endif
```

```c
// utils.c
#include "utils.h"
#include <stdlib.h>
#include <string.h>

int sum_array(const int *arr, int n) {
    int total = 0;
    for (int i = 0; i < n; i++) {
        total += arr[i];
    }
    return total;
}

void swap_int(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

char* duplicate_string(const char *s) {
    if (s == NULL) return NULL;
    char *dup = malloc(strlen(s) + 1);
    if (dup) {
        strcpy(dup, s);
    }
    return dup;
}
```
