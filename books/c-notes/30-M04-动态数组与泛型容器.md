# M04 - 动态数组与泛型容器

C 语言没有内置的动态数组，每次需要变长数组时都得手写 realloc。这篇来实现一个类似 C++ std::vector 的通用容器。

## 设计目标

- 动态扩容，自动管理内存
- 支持任意元素类型
- 尽量保证类型安全
- 用起来简单，别搞太复杂

有两种主要实现思路：void* 泛型和宏类型特化。各有利弊。

---

## 方案一：void* 泛型

### 数据结构

```c
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

typedef struct {
    void *data;           // 数据存储区
    size_t element_size;  // 单个元素大小
    size_t size;          // 当前元素数量
    size_t capacity;      // 容量
} vector;
```

核心是用 `void*` 存数据，配合 `element_size` 知道每个元素多大。存取时用 memcpy 搬运内存。

### 创建与销毁

```c
// 创建向量，指定元素大小和初始容量
vector* vec_create(size_t element_size, size_t initial_capacity) {
    vector *v = (vector*)malloc(sizeof(vector));
    if (!v) return NULL;

    v->element_size = element_size;
    v->size = 0;
    v->capacity = initial_capacity;

    if (initial_capacity > 0) {
        v->data = malloc(initial_capacity * element_size);
        if (!v->data) {
            free(v);
            return NULL;
        }
    } else {
        v->data = NULL;
    }

    return v;
}

// 销毁向量
void vec_free(vector *v) {
    if (v) {
        free(v->data);
        free(v);
    }
}
```

### 扩容逻辑

```c
// 内部函数：扩展容量
static void vec_grow(vector *v) {
    size_t new_cap = v->capacity == 0 ? 4 : v->capacity * 2;

    void *new_data = realloc(v->data, new_cap * v->element_size);
    if (!new_data) {
        // 实际项目中应该返回错误码或断言
        fprintf(stderr, "vec_grow: out of memory\n");
        return;
    }

    v->data = new_data;
    v->capacity = new_cap;
}

// 预分配容量
void vec_reserve(vector *v, size_t new_capacity) {
    if (new_capacity <= v->capacity) return;

    void *new_data = realloc(v->data, new_capacity * v->element_size);
    if (!new_data) {
        fprintf(stderr, "vec_reserve: out of memory\n");
        return;
    }

    v->data = new_data;
    v->capacity = new_capacity;
}
```

扩容策略是 2 倍增长，这是 std::vector 的做法。摊还分析下来每次 push 是 O(1)。

### 元素访问

```c
// 获取元素指针（注意：返回的是指针，不是值）
void* vec_get(vector *v, size_t index) {
    if (index >= v->size) {
        fprintf(stderr, "vec_get: index out of bounds\n");
        return NULL;
    }
    return (char*)v->data + index * v->element_size;
}

// 设置元素值
void vec_set(vector *v, size_t index, void *value) {
    if (index >= v->size) {
        fprintf(stderr, "vec_set: index out of bounds\n");
        return;
    }
    memcpy((char*)v->data + index * v->element_size, value, v->element_size);
}
```

注意 `vec_get` 返回指针。调用者需要自己转型：

```c
int *p = (int*)vec_get(v, 0);
printf("%d\n", *p);
```

### 插入与删除

```c
// 尾部添加
void vec_push(vector *v, void *value) {
    if (v->size >= v->capacity) {
        vec_grow(v);
    }
    memcpy((char*)v->data + v->size * v->element_size, value, v->element_size);
    v->size++;
}

// 尾部删除
void vec_pop(vector *v) {
    if (v->size > 0) {
        v->size--;
    }
}

// 指定位置插入
void vec_insert(vector *v, size_t index, void *value) {
    if (index > v->size) {
        fprintf(stderr, "vec_insert: index out of bounds\n");
        return;
    }

    if (v->size >= v->capacity) {
        vec_grow(v);
    }

    // 把 index 之后的元素往后挪
    if (index < v->size) {
        memmove(
            (char*)v->data + (index + 1) * v->element_size,
            (char*)v->data + index * v->element_size,
            (v->size - index) * v->element_size
        );
    }

    memcpy((char*)v->data + index * v->element_size, value, v->element_size);
    v->size++;
}

// 删除指定位置元素
void vec_erase(vector *v, size_t index) {
    if (index >= v->size) {
        fprintf(stderr, "vec_erase: index out of bounds\n");
        return;
    }

    // 把 index 之后的元素往前挪
    if (index < v->size - 1) {
        memmove(
            (char*)v->data + index * v->element_size,
            (char*)v->data + (index + 1) * v->element_size,
            (v->size - index - 1) * v->element_size
        );
    }

    v->size--;
}

// 清空
void vec_clear(vector *v) {
    v->size = 0;
}
```

### 使用示例

```c
int main(void) {
    // 创建 int 类型的向量
    vector *nums = vec_create(sizeof(int), 16);

    // push 一些值
    int val = 42;
    vec_push(nums, &val);
    val = 100;
    vec_push(nums, &val);
    val = 256;
    vec_push(nums, &val);

    // 遍历打印
    for (size_t i = 0; i < nums->size; i++) {
        int *p = (int*)vec_get(nums, i);
        printf("nums[%zu] = %d\n", i, *p);
    }

    // 修改元素
    val = 999;
    vec_set(nums, 1, &val);

    // 中间插入
    val = 77;
    vec_insert(nums, 1, &val);

    // 删除元素
    vec_erase(nums, 0);

    vec_free(nums);
    return 0;
}
```

这种方案的缺点是类型安全性差，编译器无法检查类型错误。把 `int*` 当 `double*` 用也不会报错，运行时才会出问题。

## 方案二：宏实现类型特化

用宏生成针对特定类型的向量定义，每个类型有自己的结构体和函数，类型安全得多。

### 基础宏定义

```c
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

// 定义类型特化的向量结构体和函数
#define VECTOR_DEFINE(type)                                                \
typedef struct {                                                           \
    type *data;                                                            \
    size_t size;                                                           \
    size_t capacity;                                                       \
} vector_##type;                                                           \
                                                                           \
static inline vector_##type* vector_##type##_create(size_t init_cap) {     \
    vector_##type *v = malloc(sizeof(vector_##type));                      \
    if (!v) return NULL;                                                   \
    v->size = 0;                                                           \
    v->capacity = init_cap > 0 ? init_cap : 4;                             \
    v->data = malloc(v->capacity * sizeof(type));                          \
    if (!v->data) { free(v); return NULL; }                                \
    return v;                                                              \
}                                                                          \
                                                                           \
static inline void vector_##type##_free(vector_##type *v) {                \
    if (v) { free(v->data); free(v); }                                     \
}                                                                          \
                                                                           \
static inline void vector_##type##_push(vector_##type *v, type val) {      \
    if (v->size >= v->capacity) {                                          \
        v->capacity *= 2;                                                  \
        v->data = realloc(v->data, v->capacity * sizeof(type));            \
    }                                                                      \
    v->data[v->size++] = val;                                              \
}                                                                          \
                                                                           \
static inline type vector_##type##_get(vector_##type *v, size_t idx) {     \
    return v->data[idx];                                                   \
}                                                                          \
                                                                           \
static inline void vector_##type##_set(vector_##type *v, size_t idx, type val) { \
    v->data[idx] = val;                                                    \
}                                                                          \
                                                                           \
static inline void vector_##type##_pop(vector_##type *v) {                 \
    if (v->size > 0) v->size--;                                            \
}                                                                          \
                                                                           \
static inline void vector_##type##_clear(vector_##type *v) {               \
    v->size = 0;                                                           \
}
```

### 使用特化的向量类型

```c
// 定义 int 类型的向量
VECTOR_DEFINE(int)

// 定义 double 类型的向量
VECTOR_DEFINE(double)

int main(void) {
    // 使用 int 向量
    vector_int *vi = vector_int_create(16);
    vector_int_push(vi, 10);
    vector_int_push(vi, 20);
    vector_int_push(vi, 30);

    for (size_t i = 0; i < vi->size; i++) {
        printf("vi[%zu] = %d\n", i, vector_int_get(vi, i));
    }
    vector_int_free(vi);

    // 使用 double 向量
    vector_double *vd = vector_double_create(8);
    vector_double_push(vd, 3.14);
    vector_double_push(vd, 2.718);

    for (size_t i = 0; i < vd->size; i++) {
        printf("vd[%zu] = %f\n", i, vector_double_get(vd, i));
    }
    vector_double_free(vd);

    return 0;
}
```

这个方案的好处是类型安全。`vector_int_push(vi, 3.14)` 会产生编译警告或错误。缺点是每种类型都会生成一套代码，可能导致代码膨胀。调试时宏展开的代码也不太好读。

## 方案二改进：更简洁的宏接口

如果觉得 `vector_int_push(vi, val)` 这种写法太长，可以用宏进一步包装：

```c
// 在 VECTOR_DEFINE 之后添加便捷宏
#define VEC_PUSH(v, val) _Generic((v),     \
    vector_int*: vector_int_push,          \
    vector_double*: vector_double_push     \
)(v, val)

#define VEC_GET(v, idx) _Generic((v),      \
    vector_int*: vector_int_get,           \
    vector_double*: vector_double_get      \
)(v, idx)

#define VEC_FREE(v) _Generic((v),          \
    vector_int*: vector_int_free,          \
    vector_double*: vector_double_free     \
)(v)
```

使用 C11 的 `_Generic` 让宏能根据类型自动选择正确的函数：

```c
vector_int *vi = vector_int_create(16);
VEC_PUSH(vi, 100);           // 自动调用 vector_int_push
int x = VEC_GET(vi, 0);      // 自动调用 vector_int_get
VEC_FREE(vi);                // 自动调用 vector_int_free
```

这种写法更像 C++ 的模板，但需要手动维护 `_Generic` 的类型列表。加新类型时两边都要改。

## 迭代器宏

不管用哪种方案，可以用宏简化遍历写法：

### void* 方案的迭代宏

```c
#define VEC_FOREACH(ptr, vec) \
    for (size_t _i = 0; _i < (vec)->size && ((ptr) = (typeof(ptr))vec_get(vec, _i), 1); _i++)
```

### 使用示例

```c
vector *nums = vec_create(sizeof(int), 16);
int val = 1;
for (int i = 0; i < 10; i++) {
    val = i * i;
    vec_push(nums, &val);
}

int *p;
VEC_FOREACH(p, nums) {
    printf("%d ", *p);
}
```

如果编译器不支持 `typeof`，就只能手动写：

```c
for (size_t i = 0; i < nums->size; i++) {
    int *p = (int*)vec_get(nums, i);
    printf("%d ", *p);
}
```

## 存储指针类型

如果元素是指针（比如字符串或结构体指针），要小心内存管理。vector 只存指针值，不会帮你管理指针指向的内存。

### 字符串向量示例

```c
int main(void) {
    vector *strs = vec_create(sizeof(char*), 8);

    char *s1 = strdup("hello");
    char *s2 = strdup("world");

    vec_push(strs, &s1);
    vec_push(strs, &s2);

    for (size_t i = 0; i < strs->size; i++) {
        char **p = (char**)vec_get(strs, i);
        printf("%s\n", *p);
    }

    // 记得释放字符串本身
    for (size_t i = 0; i < strs->size; i++) {
        char **p = (char**)vec_get(strs, i);
        free(*p);
    }
    vec_free(strs);

    return 0;
}
```

一种更安全的方式是 vector 保存指针数组，并提供专门的释放回调：

```c
// 带元素释放回调的向量
typedef struct {
    void *data;
    size_t element_size;
    size_t size;
    size_t capacity;
    void (*element_dtor)(void*);  // 元素析构函数
} vector_ex;

void vec_free_ex(vector_ex *v) {
    if (v) {
        if (v->element_dtor) {
            for (size_t i = 0; i < v->size; i++) {
                void *elem = (char*)v->data + i * v->element_size;
                v->element_dtor(elem);
            }
        }
        free(v->data);
        free(v);
    }
}
```

## 性能考虑

### 扩容开销

每次扩容要 realloc 和可能的内存搬运。如果知道大概需要多少元素，预分配：

```c
vector *v = vec_create(sizeof(int), 10000);  // 直接分配够用
// 或者
vec_reserve(v, 10000);  // 后续预分配
```

### push 开销

void* 方案每次 push 都要 memcpy。对于小类型（int、double）还行，大结构体会有明显开销。这时候存指针更高效：

```c
typedef struct { /* 很多字段 */ } BigStruct;

vector *v = vec_create(sizeof(BigStruct*), 16);  // 存指针
BigStruct *obj = malloc(sizeof(BigStruct));
vec_push(v, &obj);
```

### 缓存友好性

连续内存数组对缓存友好。但如果存指针，实际数据散落在堆各处，访问时缓存效率降低。这是所有指针容器的通病。

### 批量操作

如果要插入大量元素，逐个 push 效率低。可以实现批量接口：

```c
void vec_push_n(vector *v, void *values, size_t n) {
    if (v->size + n > v->capacity) {
        size_t new_cap = v->capacity;
        while (new_cap < v->size + n) {
            new_cap = new_cap == 0 ? 4 : new_cap * 2;
        }
        vec_reserve(v, new_cap);
    }
    memcpy((char*)v->data + v->size * v->element_size, values, n * v->element_size);
    v->size += n;
}
```

## 两种方案对比

| 特性 | void* 方案 | 宏类型特化 |
|------|-----------|-----------|
| 类型安全 | 运行时检查（弱） | 编译时检查（强） |
| 代码体积 | 一份实现 | 每类型一份 |
| 调试体验 | 好 | 宏展开难读 |
| 二进制兼容 | 好 | 类型耦合 |
| 适用场景 | 通用库、不确定类型 | 明确类型、性能敏感 |

实际项目中，如果类型已知且数量有限，用宏方案更安全。要是做通用库，void* 方案兼容性更好。

## 完整实现（void* 方案）

```c
// vector.h
#ifndef VECTOR_H
#define VECTOR_H

#include <stddef.h>

typedef struct {
    void *data;
    size_t element_size;
    size_t size;
    size_t capacity;
} vector;

vector* vec_create(size_t element_size, size_t initial_capacity);
void vec_free(vector *v);

void* vec_get(vector *v, size_t index);
void vec_set(vector *v, size_t index, void *value);

void vec_push(vector *v, void *value);
void vec_pop(vector *v);
void vec_insert(vector *v, size_t index, void *value);
void vec_erase(vector *v, size_t index);

void vec_clear(vector *v);
void vec_reserve(vector *v, size_t new_capacity);
void vec_shrink_to_fit(vector *v);

#endif // VECTOR_H
```

```c
// vector.c
#include "vector.h"
#include <stdlib.h>
#include <string.h>
#include <stdio.h>

vector* vec_create(size_t element_size, size_t initial_capacity) {
    vector *v = (vector*)malloc(sizeof(vector));
    if (!v) return NULL;

    v->element_size = element_size;
    v->size = 0;
    v->capacity = initial_capacity;
    v->data = NULL;

    if (initial_capacity > 0) {
        v->data = malloc(initial_capacity * element_size);
        if (!v->data) {
            free(v);
            return NULL;
        }
    }

    return v;
}

void vec_free(vector *v) {
    if (v) {
        free(v->data);
        free(v);
    }
}

void* vec_get(vector *v, size_t index) {
    if (!v || index >= v->size) return NULL;
    return (char*)v->data + index * v->element_size;
}

void vec_set(vector *v, size_t index, void *value) {
    if (!v || index >= v->size || !value) return;
    memcpy((char*)v->data + index * v->element_size, value, v->element_size);
}

static void vec_ensure_capacity(vector *v, size_t needed) {
    if (needed <= v->capacity) return;

    size_t new_cap = v->capacity == 0 ? 4 : v->capacity;
    while (new_cap < needed) {
        new_cap *= 2;
    }

    void *new_data = realloc(v->data, new_cap * v->element_size);
    if (!new_data) {
        fprintf(stderr, "vec_ensure_capacity: out of memory\n");
        return;
    }

    v->data = new_data;
    v->capacity = new_cap;
}

void vec_push(vector *v, void *value) {
    if (!v || !value) return;
    vec_ensure_capacity(v, v->size + 1);
    memcpy((char*)v->data + v->size * v->element_size, value, v->element_size);
    v->size++;
}

void vec_pop(vector *v) {
    if (v && v->size > 0) {
        v->size--;
    }
}

void vec_insert(vector *v, size_t index, void *value) {
    if (!v || !value || index > v->size) return;

    vec_ensure_capacity(v, v->size + 1);

    if (index < v->size) {
        memmove(
            (char*)v->data + (index + 1) * v->element_size,
            (char*)v->data + index * v->element_size,
            (v->size - index) * v->element_size
        );
    }

    memcpy((char*)v->data + index * v->element_size, value, v->element_size);
    v->size++;
}

void vec_erase(vector *v, size_t index) {
    if (!v || index >= v->size) return;

    if (index < v->size - 1) {
        memmove(
            (char*)v->data + index * v->element_size,
            (char*)v->data + (index + 1) * v->element_size,
            (v->size - index - 1) * v->element_size
        );
    }

    v->size--;
}

void vec_clear(vector *v) {
    if (v) v->size = 0;
}

void vec_reserve(vector *v, size_t new_capacity) {
    if (!v || new_capacity <= v->capacity) return;

    void *new_data = realloc(v->data, new_capacity * v->element_size);
    if (!new_data) {
        fprintf(stderr, "vec_reserve: out of memory\n");
        return;
    }

    v->data = new_data;
    v->capacity = new_capacity;
}

void vec_shrink_to_fit(vector *v) {
    if (!v || v->size == v->capacity) return;

    void *new_data = realloc(v->data, v->size * v->element_size);
    if (new_data) {
        v->data = new_data;
        v->capacity = v->size;
    }
}
```

## 小结

动态数组是 C 项目中最常用的数据结构之一。两种实现思路各有适用场景：

- **void* 方案**适合通用库，一份代码处理所有类型，但类型安全靠开发者自觉
- **宏特化方案**适合明确类型的场景，编译期检查类型，代价是代码膨胀

不管哪种方案，核心都是 `realloc + memcpy/memmove` 这套内存操作。理解这两点，就能根据实际需求灵活调整。
