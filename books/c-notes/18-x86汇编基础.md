# 第18章 x86 汇编基础

汇编语言是机器指令的人可读形式。理解汇编有助于深入理解程序的底层行为，也是调试和逆向分析的基础。

## 寄存器组

### 通用寄存器

x86-32 位架构有 8 个通用寄存器：

| 寄存器 | 名称 | 常见用途 |
|--------|------|----------|
| EAX | 累加器 | 算术运算、函数返回值 |
| EBX | 基址寄存器 | 通用、曾用于 PIC |
| ECX | 计数器 | 循环计数、字符串操作 |
| EDX | 数据寄存器 | 乘除法、I/O 操作 |
| ESI | 源索引 | 字符串/数组操作源地址 |
| EDI | 目的索引 | 字符串/数组操作目的地址 |
| EBP | 基址指针 | 栈帧基址 |
| ESP | 栈指针 | 栈顶指针 |

x86-64 扩展为 16 个通用寄存器：RAX、RBX、RCX、RDX、RSI、RDI、RBP、RSP、R8-R15。

### 寄存器子集

可以访问寄存器的部分位：

```
64位: RAX
32位: EAX
16位: AX
8位:  AL(低8位) / AH(高8位)
```

```asm
mov rax, 0x1122334455667788
mov eax, 0xFFFFFFFF      ; 同时清零高32位
mov ax, 0x1234
mov al, 0x56
```

### 特殊寄存器

**EIP/RIP**：指令指针，存放下一条要执行的指令地址。不能直接修改，只能通过跳转指令改变。

**EFLAGS/RFLAGS**：状态寄存器，存储条件码：
- ZF（Zero Flag）：结果为零时置 1
- SF（Sign Flag）：结果为负时置 1
- CF（Carry Flag）：无符号溢出时置 1
- OF（Overflow Flag）：有符号溢出时置 1

### 段寄存器

x86 架构有 CS、DS、SS、ES、FS、GS 段寄存器，现代操作系统使用平坦内存模型，段寄存器作用已弱化。

## 数据传送指令

### MOV

最基本的数据传送指令。

```asm
mov dest, src    ; dest = src

; 立即数到寄存器
mov eax, 42

; 寄存器到寄存器
mov eax, ebx

; 内存到寄存器
mov eax, [ebx]      ; ebx 指向的内存值
mov eax, [ebx+4]    ; ebx+4 处的值
mov eax, [ebx+ecx*4]; 带比例因子的寻址

; 寄存器到内存
mov [ebx], eax

; 立即数到内存
mov dword [ebx], 42
```

注意：x86 不支持内存到内存的直接传送，需要通过寄存器中转。

```asm
; 错误
mov [ebx], [ecx]

; 正确
mov eax, [ecx]
mov [ebx], eax
```

### LEA

加载有效地址，计算地址而不是取值。

```asm
; 假设 ebx = 0x1000
lea eax, [ebx+8]     ; eax = 0x1008
mov eax, [ebx+8]     ; eax = 内存 0x1008 处的值

; 常用于计算
lea eax, [ebx+ecx*4] ; eax = ebx + ecx*4（乘法优化）
```

### PUSH/POP

栈操作指令。

```asm
push eax    ; esp -= 4, [esp] = eax
pop eax     ; eax = [esp], esp += 4

; 保存和恢复寄存器
push ebx
push ecx
; ... 使用 ebx, ecx ...
pop ecx
pop ebx
```

## 算术运算指令

### ADD/SUB

加法减法指令。

```asm
add eax, 10     ; eax = eax + 10
add eax, ebx    ; eax = eax + ebx
add [ebx], eax  ; 内存值 += eax

sub eax, 10     ; eax = eax - 10
```

### INC/DEC

自增自减指令，不影响进位标志。

```asm
inc eax     ; eax++
dec ebx     ; ebx--
```

### IMUL/IDIV

有符号乘法除法。

```asm
; 单操作数形式
imul ebx            ; edx:eax = eax * ebx

; 双操作数形式
imul eax, ebx       ; eax = eax * ebx
imul eax, ebx, 10   ; eax = ebx * 10

; 除法：edx:eax / 操作数 -> eax(商), edx(余)
mov eax, 100
mov edx, 0          ; 必须先清零 edx
idiv dword [divisor] ; eax = 商, edx = 余
```

无符号乘除使用 MUL/DIV 指令。

## 逻辑运算指令

```asm
and eax, 0x0F   ; eax &= 0x0F
or  eax, 0x80   ; eax |= 0x80
xor eax, eax    ; eax = 0（常用清零方式）
not eax         ; eax = ~eax
```

TEST 指令执行 AND 操作但只设置标志，不保存结果：

```asm
test eax, eax   ; 检查 eax 是否为零
jz  is_zero     ; 如果 ZF=1（结果为零）则跳转
```

## 控制转移指令

### 无条件跳转

```asm
jmp label       ; 直接跳转
jmp eax         ; 间接跳转到 eax 存储的地址
jmp [ebx]       ; 间接跳转到内存存储的地址
```

### 条件跳转

基于 EFLAGS 状态跳转：

| 指令 | 条件 | 含义 |
|------|------|------|
| JZ/JE | ZF=1 | 为零/相等 |
| JNZ/JNE | ZF=0 | 非零/不相等 |
| JG/JNLE | ZF=0 && SF=OF | 大于（有符号） |
| JGE/JNL | SF=OF | 大于等于（有符号） |
| JL/JNGE | SF≠OF | 小于（有符号） |
| JLE/JNG | ZF=1 \|\| SF≠OF | 小于等于（有符号） |
| JA/JNBE | CF=0 && ZF=0 | 大于（无符号） |
| JB/JNAE | CF=1 | 小于（无符号） |

```asm
cmp eax, ebx    ; eax - ebx，设置标志位
jg  greater     ; 如果 eax > ebx（有符号比较）则跳转
jl  less        ; 如果 eax < ebx 则跳转
je  equal       ; 如果 eax == ebx 则跳转
```

### 循环结构

```asm
; C: for (int i = 0; i < 10; i++) { ... }
    mov ecx, 10
loop_start:
    ; 循环体
    dec ecx
    jnz loop_start
```

LOOP 指令使用 ECX 作为计数器：

```asm
    mov ecx, 10
loop_start:
    ; 循环体
    loop loop_start    ; ecx--; if (ecx != 0) goto loop_start
```

## 函数调用约定

### CALL/RET

```asm
call func   ; push eip; jmp func
ret         ; pop eip（返回到调用处）
ret 8       ; 返回并弹出 8 字节参数（stdcall）
```

### cdecl（C 调用约定）

x86-32 Linux/Windows 默认约定：

- 参数从右到左入栈
- 调用者清理栈
- 返回值在 EAX

```c
int add(int a, int b) {
    return a + b;
}

// 调用 add(1, 2)
```

```asm
; 调用方
push 2          ; 参数 b
push 1          ; 参数 a
call add
add esp, 8      ; 调用者清理栈

; 被调用方
add:
    push ebp
    mov ebp, esp
    mov eax, [ebp+8]   ; 参数 a
    add eax, [ebp+12]  ; 参数 b
    pop ebp
    ret
```

### stdcall

Windows API 使用的约定：

- 参数从右到左入栈
- 被调用者清理栈（RET n）
- 返回值在 EAX

```asm
; 被调用方
add:
    push ebp
    mov ebp, esp
    mov eax, [ebp+8]
    add eax, [ebp+12]
    pop ebp
    ret 8           ; 返回并弹出 8 字节
```

### x86-64 调用约定（System V AMD64 ABI）

Linux/macOS 使用：

- 前 6 个整数参数：RDI、RSI、RDX、RCX、R8、R9
- 额外参数入栈
- 返回值在 RAX
- 调用者清理栈

```c
int add(int a, int b, int c, int d, int e, int f, int g) {
    return a + b + c + d + e + f + g;
}
```

```asm
; 调用方
mov edi, 1      ; a
mov esi, 2      ; b
mov edx, 3      ; c
mov ecx, 4      ; d
mov r8d, 5      ; e
mov r9d, 6      ; f
push 7          ; g（第 7 个参数入栈）
call add
add rsp, 8      ; 清理栈

; 被调用方
add:
    mov eax, edi
    add eax, esi
    add eax, edx
    add eax, ecx
    add eax, r8d
    add eax, r9d
    add eax, [rsp+8]  ; 第 7 个参数
    ret
```

Windows x64 使用不同的约定：前 4 个参数通过 RCX、RDX、R8、R9 传递。

## 完整示例

### C 代码

```c
int sum(int n) {
    int result = 0;
    for (int i = 1; i <= n; i++) {
        result += i;
    }
    return result;
}

int main() {
    int s = sum(100);
    return s;
}
```

### 对应汇编（x86-32）

```asm
sum:
    push ebp
    mov ebp, esp
    sub esp, 8          ; 局部变量空间

    mov dword [ebp-4], 0; result = 0
    mov dword [ebp-8], 1; i = 1

.loop:
    mov eax, [ebp-8]    ; i
    cmp eax, [ebp+8]    ; i <= n ?
    jg .end

    mov eax, [ebp-4]
    add eax, [ebp-8]    ; result += i
    mov [ebp-4], eax

    inc dword [ebp-8]   ; i++
    jmp .loop

.end:
    mov eax, [ebp-4]    ; 返回值
    mov esp, ebp
    pop ebp
    ret

main:
    push ebp
    mov ebp, esp

    push 100
    call sum
    add esp, 4

    mov esp, ebp
    pop ebp
    ret
```
