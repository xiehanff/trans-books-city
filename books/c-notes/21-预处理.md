# 第21章 预处理

预处理在编译之前执行，处理以 `#` 开头的指令。理解预处理机制有助于编写灵活高效的代码。

## #include 指令

`#include` 将指定文件的内容插入到当前位置。

### 两种形式

```c
#include <stdio.h>    // 系统头文件，搜索系统目录
#include "myheader.h" // 用户头文件，先搜索当前目录，再搜索系统目录
```

### 头文件保护

防止头文件被重复包含：

```c
// myheader.h
#ifndef MYHEADER_H
#define MYHEADER_H

// 头文件内容

#endif
```

更简洁的方式（非标准但广泛支持）：

```c
// myheader.h
#pragma once

// 头文件内容
```

两种方式可以同时使用以确保兼容性。

### 常见用法

```c
#include <stdio.h>     // 标准输入输出
#include <stdlib.h>    // 标准库函数
#include <string.h>    // 字符串操作
#include <stdint.h>    // 固定宽度整数类型
#include <stdbool.h>   // 布尔类型（C99）
```

### 包含路径

```bash
# 添加包含路径
gcc -I/path/to/headers main.c

# 查看预处理结果
gcc -E main.c
```

## #define 宏定义

### 常量定义

```c
#define PI 3.14159
#define MAX_SIZE 100
#define BUFFER_SIZE (1024 * 1024)  // 建议加括号
```

注意：宏定义中的表达式要加括号，避免展开后优先级问题。

```c
#define DOUBLE(x) x * 2    // 错误：DOUBLE(1+2) 展开为 1+2 * 2 = 5
#define DOUBLE(x) ((x) * 2) // 正确：DOUBLE(1+2) 展开为 ((1+2) * 2) = 6
```

### 宏函数

```c
#define MAX(a, b) ((a) > (b) ? (a) : (b))
#define MIN(a, b) ((a) < (b) ? (a) : (b))
#define ABS(x) ((x) >= 0 ? (x) : -(x))

int x = MAX(3, 5);         // ((3) > (5) ? (3) : (5))
int y = MAX(a++, b);       // 危险！a 可能自增两次
```

宏函数的副作用问题是宏的重要陷阱。

### 字符串化和连接

`#` 将参数转为字符串：

```c
#define STR(x) #x
#define XSTR(x) STR(x)

STR(hello)       // "hello"
XSTR(hello)      // "hello"

#define VALUE 100
STR(VALUE)        // "VALUE"
XSTR(VALUE)       // "100"（先展开宏，再字符串化）
```

`##` 连接两个标记：

```c
#define CONCAT(a, b) a##b
#define MAKE_FUNC(name) void func_##name() {}

CONCAT(abc, def)  // abcdef
MAKE_FUNC(test)   // void func_test() {}
```

### 多行宏

使用反斜杠续行：

```c
#define SWAP(a, b, type) \
    do { \
        type temp = a; \
        a = b; \
        b = temp; \
    } while(0)

// 使用
SWAP(x, y, int);
```

使用 `do { ... } while(0)` 确保宏在使用时像一条语句。

### 取消宏

```c
#define DEBUG 1
// ...
#undef DEBUG
```

## 宏函数与内联函数

### 宏函数的问题

```c
#define SQUARE(x) ((x) * (x))

int a = 5;
int b = SQUARE(a++);  // 展开为 ((a++) * (a++))，a 自增两次！
```

### 内联函数

```c
inline int square(int x) {
    return x * x;
}

int a = 5;
int b = square(a++);  // 正确！a 只自增一次
```

内联函数的优点：
- 类型检查
- 只求一次参数值
- 调试时可断点

宏的优点：
- 无类型限制
- 零开销（无函数调用）
- 可用于泛型编程

### 泛型宏（C11）

```c
#define print_type(x) _Generic((x), \
    int: print_int, \
    float: print_float, \
    char*: print_string \
)(x)

void print_int(int x) { printf("int: %d\n", x); }
void print_float(float x) { printf("float: %f\n", x); }
void print_string(char* x) { printf("string: %s\n", x); }

print_type(42);       // int: 42
print_type(3.14f);    // float: 3.140000
print_type("hello");  // string: hello
```

## 条件编译

### #ifdef / #ifndef

```c
#ifdef DEBUG
    printf("Debug mode\n");
#endif

#ifndef PI
    #define PI 3.14159
#endif
```

### #if / #elif / #else

```c
#if VERSION == 1
    #define FEATURE_A 1
#elif VERSION == 2
    #define FEATURE_A 1
    #define FEATURE_B 1
#else
    #define FEATURE_A 0
    #define FEATURE_B 0
#endif
```

### defined 运算符

```c
#if defined(DEBUG) && !defined(RELEASE)
    // 调试代码
#endif

// 组合多个条件
#if defined(PLATFORM_LINUX) || defined(PLATFORM_MACOS)
    // Unix-like 平台代码
#endif
```

### 平台检测

```c
#ifdef _WIN32
    // Windows 代码
#elif defined(__linux__)
    // Linux 代码
#elif defined(__APPLE__)
    // macOS 代码
#endif
```

### 调试开关

```c
#define DEBUG 1

#if DEBUG
    #define LOG(msg) printf("[DEBUG] %s:%d: %s\n", __FILE__, __LINE__, msg)
#else
    #define LOG(msg)
#endif

LOG("程序启动");  // DEBUG=1 时输出，否则什么都不做
```

编译时通过 `-D` 选项定义宏：

```bash
gcc -DDEBUG=1 main.c     # 定义 DEBUG 为 1
gcc -DDEBUG main.c       # 定义 DEBUG 为 1（默认）
gcc -DVERSION=2 main.c   # 定义 VERSION 为 2
```

## #pragma 指令

`#pragma` 提供编译器特定的指令。

### 常用 pragma

```c
#pragma once  // 头文件只包含一次

#pragma pack(push, 1)  // 1 字节对齐
struct Packed {
    char a;
    int b;
};
#pragma pack(pop)  // 恢复默认对齐

#pragma warning(disable: 4996)  // 禁用特定警告（MSVC）

#pragma GCC diagnostic ignored "-Wunused-variable"  // GCC/Clang
```

### ## 运算符在 pragma 中

```c
#define PRAGMA(x) _Pragma(#x)
PRAGMA(message "编译到这里了")
```

## 预定义宏

编译器内置的宏定义。

### 标准预定义宏

```c
__FILE__       // 当前源文件名（字符串）
__LINE__       // 当前行号（整数）
__func__       // 当前函数名（C99，字符串）
__DATE__       // 编译日期（字符串）
__TIME__       // 编译时间（字符串）
__STDC__       // 是否符合 ANSI C
__STDC_VERSION__  // C 标准版本（如 199901L 表示 C99）
```

### 使用示例

```c
void log_error(const char* msg) {
    fprintf(stderr, "[%s:%d] %s: %s\n",
            __FILE__, __LINE__, __func__, msg);
}

int main() {
    log_error("发生了错误");
    return 0;
}
```

输出：
```
[main.c:8] main: 发生了错误
```

### 条件判断示例

```c
#if __STDC_VERSION__ >= 201112L
    // C11 特性
    #define THREAD_LOCAL _Thread_local
#elif __STDC_VERSION__ >= 199901L
    // C99 特性
#else
    // C89/C90
#endif
```

### 编译器特定宏

```c
#ifdef __GNUC__
    // GCC 特定代码
    #define ATTR_UNUSED __attribute__((unused))
#elif defined(_MSC_VER)
    // MSVC 特定代码
    #define ATTR_UNUSED
#endif
```

## 常见陷阱

### 宏定义末尾加分号

```c
#define VALUE 100;  // 错误

int x = VALUE;  // 展开为 int x = 100;; 多了分号
```

### 宏参数没有括号

```c
#define DOUBLE(x) x * 2  // 错误

int y = DOUBLE(1 + 2);  // 展开为 1 + 2 * 2 = 5，不是 6
```

### 宏参数有副作用

```c
#define MAX(a, b) ((a) > (b) ? (a) : (b))

int x = 1, y = 2;
int z = MAX(x++, y);  // x 可能自增 1 或 2 次
```

### 宏定义中的空格

```c
#define F (x) ((x) * 2)  // F 展开为 (x) ((x) * 2)
#define F(x) ((x) * 2)   // F(x) 展开为 ((x) * 2)
```

## 断言宏

```c
#include <assert.h>

void func(int* p) {
    assert(p != NULL);  // 如果 p 为 NULL，程序终止并输出错误信息
    // 使用 p
}
```

自定义断言：

```c
#ifdef DEBUG
    #define ASSERT(expr) \
        do { \
            if (!(expr)) { \
                fprintf(stderr, "Assertion failed: %s, %s:%d\n", \
                        #expr, __FILE__, __LINE__); \
                abort(); \
            } \
        } while(0)
#else
    #define ASSERT(expr)
#endif
```
